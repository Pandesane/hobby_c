defmodule HealthierWeb.RoomChannel do
  alias HealthierWeb.Endpoint
  alias Healthier.MessageContext
  # alias Healthier.UserContext
  alias Processes.File.FileUploadAgent

  use HealthierWeb, :channel

  @new_message_sent_topic "new_message"
  @new_attached_file "new_attached_file"
  @file_upload_topic "file-upload-done"
  @user_typing "user_typing"
  @user_stopped_typing "user_stopped_typing"
  intercept [@new_message_sent_topic, @new_attached_file, @user_typing]
  # intercept [@new_message_sent_topic, @new_attached_file]

  @impl true
  def join("room:", payload, socket) do
    case authorized?(payload, socket) do
      {true, socket} ->
        # Endpoint.subscribe(@new_message_sent_topic)
        {:ok, socket}

      {false, _socket} ->
        {:error, %{reason: "unauthorized"}}
    end
  end

  # Channels can be used in a request/response fashion
  # by sending replies to requests from the client
  @impl true
  def handle_in("ping", payload, socket) do
    {:reply, {:ok, payload}, socket}
  end

  def handle_in(
        "user_typing",
        %{"receiver_id" => receiver_id, "sender_id" => sender_id} = payload,
        socket
      ) do
    socket
    |> broadcast(@user_typing, %{
      receiver_id: receiver_id,
      sender_id: sender_id
    })

    {:noreply, socket}
  end

  # It is also common to receive messages from the client and
  # broadcast to everyone in the current topic (room:lobby).
  @impl true
  def handle_in(
        "message",
        %{"receiver_id" => receiver_id, "sender_id" => sender_id} = payload,
        socket
      ) do
    {:ok, msg} = MessageContext.create_message(payload)
    message = MessageContext.format_message(msg)
    dbg(message)

    socket
    |> broadcast(@new_message_sent_topic, message)

    push(socket, "new_message", message)

    {:noreply, socket}
  end




  # {:file_upload_done, "f35b01eb-d16f-4e62-9fde-85392bf113d8", :file,
  #  %{
  #    file_type: "image/jpeg",
  #    file_path: "http://localhost:4000/videos/IMG_20240105_094444_002.jpg"
  #  }}


  defp format_chats(_chats = [], formatted_chats) do
    dbg(formatted_chats)
    formatted_chats
  end

  defp format_chats([chat | rest_chats] = _chats, formatted_chats) do
    new_chat = %{name: chat.name, id: chat.id, email: chat.email}
    new_formatted_chats = [new_chat | formatted_chats]
    format_chats(rest_chats, new_formatted_chats)
  end
end
